# 安裝用 powershell

[Console]::OutputEncoding = [System.Text.Encoding]::UTF8

Write-Host "這是一個 Discord AI agent 專案。"
Write-Host "你現在應該是在安裝此專案吧。"
Write-Host "請先安裝 Python 和 git 這兩個前置作業。"

5..1 | ForEach-Object {
    Write-Host "`r將在 $_ 秒後開始安裝..." -NoNewline
    Start-Sleep -Seconds 1
}

Write-Host "`r將在 0 秒後開始安裝..."

Start-Sleep -Seconds 1

$ProgressPreference = 'SilentlyContinue'

function Test-Command($name) {
    return [bool](Get-Command $name -ErrorAction SilentlyContinue)
}

function Update-SessionPath {
    $machinePath = [Environment]::GetEnvironmentVariable("Path", "Machine")
    $userPath    = [Environment]::GetEnvironmentVariable("Path", "User")
    $env:Path    = "$machinePath;$userPath"
}

function Install-Python {
    Write-Host "正在下載 Python 3.12.8..." -ForegroundColor Cyan
    Invoke-WebRequest "https://www.python.org/ftp/python/3.12.8/python-3.12.8-amd64.exe" -OutFile ".\install-python.exe"

    Write-Host "正在安裝 Python 3.12.8..." -ForegroundColor Yellow
    Start-Process -Wait -FilePath ".\install-python.exe" -ArgumentList "/passive PrependPath=1 Include_launcher=1"

    Remove-Item ".\install-python.exe" -ErrorAction SilentlyContinue
    Update-SessionPath
}

function Install-Git {
    Write-Host "正在下載 Git 2.56.0.2..." -ForegroundColor Cyan
    Invoke-WebRequest "https://github.com/git-for-windows/git/releases/download/v2.56.0.windows.2/Git-2.56.0.2-64-bit.exe" -OutFile ".\install-git.exe"

    Write-Host "正在安裝 Git 2.56.0.2..." -ForegroundColor Yellow
    Start-Process -Wait -FilePath ".\install-git.exe" -ArgumentList "/SILENT /NORESTART"

    Remove-Item ".\install-git.exe" -ErrorAction SilentlyContinue
    Update-SessionPath
}

if (-not (Test-Command "python")) {
    Write-Host "【提示】未檢測到 Python" -ForegroundColor Red
    $answer = Read-Host "是否要自動安裝 Python？ (y/n)"
    if ($answer -eq 'y' -or $answer -eq 'Y') {
        Install-Python
    } else {
        Write-Host "已略過 Python 安裝，請自行安裝後再執行此腳本。" -ForegroundColor Yellow
        exit
    }
}

if (-not (Test-Command "git")) {
    Write-Host "【提示】未檢測到 Git" -ForegroundColor Red
    $answer = Read-Host "是否要自動安裝 Git？ (y/n)"
    if ($answer -eq 'y' -or $answer -eq 'Y') {
        Install-Git
    } else {
        Write-Host "已略過 Git 安裝，請自行安裝後再執行此腳本。" -ForegroundColor Yellow
        exit
    }
}

if ((Test-Command "python") -and (Test-Command "git")) {
    Write-Host "Python 與 Git 已成功安裝完成！" -ForegroundColor Green
} else {
    Write-Host "安裝後仍無法偵測到部分工具，請重新開啟 PowerShell 後再試。" -ForegroundColor Red
}

$confirm = Read-Host "你是否要新增 Discord AI agent 到你的專案？ (y/n)"

if ($confirm.ToLower() -ne "y") {
    Write-Host "操作已終止。" -ForegroundColor Yellow
    exit
}

$projectName = Read-Host "請輸入資料夾名稱"

if ([string]::IsNullOrWhiteSpace($projectName)) {
    Write-Host "專案名稱不能為空，操作已終止。" -ForegroundColor Red
    exit
}

New-Item -ItemType Directory -Path $projectName -Force | Out-Null
Set-Location -Path $projectName

Write-Host "正在自動建立專案..."

git clone -q https://github.com/brndan918/discord-AI-agent.git .

if (Test-Path ".env") {
    Remove-Item -Path ".env.example" -ErrorAction SilentlyContinue
} else {
    Rename-Item -Path ".env.example" -NewName ".env" -ErrorAction SilentlyContinue
}

Remove-Item -Path "install_ai-agent.txt" -Force | Out-Null

Write-Host "專案建立完成！" -ForegroundColor Green

$esc = [char]27
$url = "https://github.com/brndan918/discord-AI-agent#dai-discord-ai-tool-%E5%AE%89%E8%A3%9D%E8%88%87%E4%BD%BF%E7%94%A8%E6%8C%87%E5%8D%97"
$link = "${esc}]8;;${url}${esc}\README.md${esc}]8;;${esc}\"

Write-Host "請查看 $link"
Write-Host "請運行 install_requirements.bat 安裝依賴"

Start-Sleep -Seconds 2

explorer .
